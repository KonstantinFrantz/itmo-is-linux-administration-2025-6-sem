#!/bin/bash

PID=$1
GROUP_NAME="dynamic_process_$PID"

cgcreate -g cpu:/$GROUP_NAME

echo $PID >> /sys/fs/cgroup/$GROUP_NAME/cgroup.procs

while true; do
    CPU_LOAD=$(top -bn1 | grep "Cpu(s)" | awk '{print $2}' | cut -d'.' -f1)
    
    if [ $CPU_LOAD -lt 20 ]; then
        cgset -r cpu.max="80000 100000" $GROUP_NAME
    elif [ $CPU_LOAD -gt 60 ]; then
        cgset -r cpu.max="30000 100000" $GROUP_NAME
    fi
    
    sleep 5
done

#!/bin/bash

LOWER_DIR=~/overlay_/lower
UPPER_DIR=~/overlay_/upper
WORK_DIR=~/overlay_/work
MERGED_DIR=~/overlay_/merged
AUDIT_LOG=56_audit.log

> ${AUDIT_LOG}

find ${UPPER_DIR} -type c -o -name ".wh.*" | while read file; do
    echo "Found: ${file}" >> ${AUDIT_LOG}
done

cd ${LOWER_DIR}
find . -type f -o -type d | while read file; do
    if [ "$file" != "." ]; then
        rel_path=${file#./}

        if [ ! -e "${MERGED_DIR}/${rel_path}" ]; then
            echo "Missing in merged: ${rel_path}" >> ${AUDIT_LOG}
        elif [ -f "${LOWER_DIR}/${rel_path}" ] && [ -f "${MERGED_DIR}/${rel_path}" ]; then
            if ! cmp -s "${LOWER_DIR}/${rel_path}" "${MERGED_DIR}/${rel_path}"; then
                echo "Content mismatch: ${rel_path}" >> ${AUDIT_LOG}
            fi
        fi
    fi
done
